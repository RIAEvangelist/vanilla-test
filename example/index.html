Core pattern

Catch assertion errors, then close the test.

const test = new VanillaTest();

test.expects('the result is a number');
try {
    test.is.number(1 + 2);
    test.pass();
} catch (error) {
    console.error(error);
    test.fail();
}
test.done();

const result = test.report();
